it("считает U+200B и смесь U+200B, табуляции и NBSP пустым названием", () => {
    const task = { id: 1, title: "Купить хлеб", done: false }
    for (const blank of ["\u200B", "\t\u200B\u00A0"]) {
        const saved = [task, { id: 2, title: blank, done: false }]
        expect.soft(createTask(2, blank)).toBeUndefined()
        expect.soft(renameTask([task], 1, blank)).toEqual([task])
        expect.soft(visibleTasks(saved, "all")).toEqual([task])
        expect.soft(openCount(saved)).toBe(1)
    }
    expect(renameTask([task], 1, "\tНовое имя\u00A0")[0].title).toBe("Новое имя")

})
